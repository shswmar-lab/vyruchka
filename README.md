# ============================================================
#  ВЫРУЧКА ВОДИТЕЛЕЙ — клиент-серверная версия
# ============================================================
#
#  Бывший standalone-HTML переделан в веб-приложение:
#    - сервер: Python + FastAPI + SQLite (Ubuntu, native: systemd + nginx + venv)
#    - клиент: браузер (SPA из одного index.html)
#    - авторизация: JWT в httpOnly-cookie, пароли bcrypt
#    - роли: Администратор / Пользователь / Руководитель
#      (Руководитель — только «Аналитика» и «Графики», запись и админ-разделы закрыты)
#    - данные выручки — общая база для всех пользователей
#    - админ-панель: добавление / редактирование / удаление пользователей
#
# ============================================================

## Структура проекта
#
# app/                  # серверный код FastAPI
#   main.py             # приложение, статика, CSRF-защита
#   db.py               # SQLite (создание таблиц автоматически)
#   schemas.py          # Pydantic-модели
#   auth.py             # bcrypt + JWT + зависимости get_current_user / require_admin
#   create_admin.py     # создание первичного администратора
#   routers/            # auth, users, rows, drivers, routes, audit
# static/               # index.html + xlsx.full.min.js (раздаёт сервер/nginx)
# deploy/               # systemd-юнит, nginx.conf, deploy.sh
# test/                 # pytest: auth, роли, CRUD, дедупликация, экспорт, журнал
# .env.example          # шаблон конфигурации (секретов здесь нет)
# requirements.txt

## Предварительные требования (Ubuntu 22.04/24.04)
#   sudo apt update && sudo apt install -y python3 python3-venv nginx openssl

## УСТАНОВКА (запускать от root на сервере)
#
# 1. Залить проект на сервер, например в /opt:
#      rsync -av ./ root@host:/root/vyruchka   (или git clone)
#
# 2. Запустить установку:
#      cd /root/vyruchka
#      sudo bash deploy/deploy.sh
#
#    Скрипт:
#      - создаёт системного пользователя `vyruchka`
#      - копирует код в /opt/vyruchka
#      - ставит зависимости в venv
#      - генерирует .env с SECRET_KEY и ОСТАНАВЛИВАЕТСЯ, чтобы вы задали пароль админа
#
# 3. Отредактировать конфигурацию:
#      sudo nano /opt/vyruchka/.env
#        ADMIN_USERNAME=admin
#        ADMIN_PASSWORD=ВАШ_НАДЁЖНЫЙ_ПАРОЛЬ
#        COOKIE_SECURE=true        # если будет HTTPS
#
# 4. Повторно запустить установку (досоздаст админа, systemd, nginx):
#      cd /root/vyruchka
#      sudo bash deploy/deploy.sh
#
# 5. (Рекомендуется) Выпустить HTTPS-сертификат:
#      sudo apt install -y certbot python3-certbot-nginx
#      sudo certbot --nginx -d ВАШ_ДОМЕН
#      (в .env выставить COOKIE_SECURE=true и перезапустить сервис)
#
# Проверка:
#   systemctl status vyruchka
#   curl -I http://localhost/
#   journalctl -u vyruchka -f

## РАЗРАБОТКА / ЛОКАЛЬНЫЙ ЗАПУСК (Windows или Linux)
#   python -m venv .venv
#   .venv/Scripts/activate           (Windows)  |  source .venv/bin/activate  (Linux/macOS)
#   pip install -r requirements.txt
#   set DB_PATH=data/dev.db & set SECRET_KEY=dev                (Windows)
#   python -m app.create_admin      # создаст admin-админа
#   uvicorn app.main:app --reload --port 8000
#   Открыть http://localhost:8000   (логин: admin / пароль из консоли)

## ПЕРВАЯ АВТОРИЗАЦИЯ
#   Логин и пароль берутся из .env (ADMIN_USERNAME / ADMIN_PASSWORD); deploy.sh
#   подгружает .env при создании администратора.
#   Если ADMIN_PASSWORD пуст — пароль генерируется и печатается в консоль
#   (при create_admin.py). Смените его в панели «Администрирование».
#
#   Ручное создание админа на сервере (только из каталога приложения):
#     sudo -u vyruchka bash -c 'cd /opt/vyruchka && exec ./venv/bin/python -m app.create_admin'

## ПЕРЕНОС ДАННЫХ ИЗ СТАРОЙ ВЕРСИИ
#   1. В старом приложении «Выручка.html» нажмите «💾 Excel» — выгрузите все записи.
#   2. В новом приложении, войдя администратором, загрузите файл в шапке: «📥 Excel»,
#      выбрав режим «Заменить» (или «Дополнить» — добавятся только новые строки).
#   3. Справочники водителей и маршрутов загрузите файлами: «📂 Выбрать».

## РОЛИ И ПРАВА
#   ─ Администратор: всё + раздел «Администрирование» (CRUD пользователей, «Журнал
#     действий») + «Очистить всё» + импорт Excel + «Все (CSV)».
#   ─ Пользователь:  ввод/редактирование записей, сводные, экспорт текущего Excel, печать,
#                    просмотр шапки «Аналитики»; без импорта и экспорта всех данных.
#   ─ Руководитель:  только «Аналитика» и «Графики» (с экспортом/печатью сводных);
#                    запись записей, справочники, экспорт CSV и админ-разделы закрыты.
#
#   Ограничения безопасности:
#   - запрет удаления/блокировки/понижения СВОЕГО аккаунта;
#   - «Очистить всё», импорт (Заменить/Дополнить), экспорт всех (CSV) и журнал действий
#     доступны только администратору (проверяется на сервере);
#   - CSRF-защита: кросс-доменные мутирующие запросы блокируются;
#   - пароль хранится только как bcrypt-хэш, кука httpOnly+SameSite=Lax.

## ФУНКЦИОНАЛ
#   - Таблица «Введённые данные» с периодом (месяц/год) и фильтрами по ФИО/дате/автору;
#   - вкладка «Аналитика»: сводные по водителям/датам/маршрутам (суммы) с детализацией,
#     история «все время»;
#   - вкладка «Графики»: годовой, месячный и произвольный по дням
#     (строятся по нажатию «Построить», с итоговой суммой);
#   - импорт из Excel двумя режимами (Дополнить / Заменить, только админ);
#     при нахождении дублей — запрос «импортировать их вместе или нет»,
#     затем предложение сохранить дубли в CSV;
#   - экспорт: текущие записи в Excel; все записи «Все (CSV)» (только админ);
#   - предупреждение о возможном дубле при вводе (ФИО + дата + маршрут);
#   - журнал действий администратора (импорт, очистка, управление пользователями).

## АРХИТЕКТУРА КЛЮЧЕВЫХ МОМЕНТОВ
#   - Каждый запрос открывает собственное соединение с SQLite (надёжно для этого объёма).
#   - JWT (7 дней) кладётся в httpOnly-cookie — фронтенду не нужно хранить токен в JS.
#   - Кнопка/вкладка админ-панели скрыта для обычных пользователей, но все проверки
#     ролей продублированы на сервере.
#   - Импорт/экспорт Excel выполняются на клиенте (sheet.js), сервер получает готовый JSON;
#     экспорт «Все (CSV)» и режим «Дополнить» (дедуп) — на сервере.
